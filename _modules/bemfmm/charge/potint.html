def potint(r1, r2, r3, normal, ObsPoint):
    """
    This function computes single potential integrals I = Is(1/r) and
    integrals IRho=Is(vec(r)/r), respectively,  and
    follows the method described in: Wilton DR, Rao SM, Glisson AW,
    Schaubert DH, Al-Bundak OM, Butler CM. Potential integrals for uniform
    and linear source distribution on polygonal and polyhedral domains.
    IEEE Trans Antennas Propag 1984;AP-32 (3):276–281 - uses the last
    formula in Eq. (5) on page 279.
    Inputs:
    r1, r2, r3 - three triangle vertices (1x3 rows)
    normal - unit normal vector to the triangle:
              tempv           = cross(r2-r1, r3-r1);
              temps           = sqrt(tempv(1)^2 + tempv(2)^2 + tempv(3)^2);
              normal          = tempv'/temps;
    ObsPoint - Nx3 array of observation points
    Outputs:
    I - Nx1 array of integrals I = Is(1/r)
    IRho - Nx3 array of integrals IRho = Is(vec(r)/r)
    To obtain the complete integrals these results should be multiplied by
    triangle area A

    Copyright SNM 2002-2020

    SP 2026
    """

    N = ObsPoint.shape[0]

    Z = np.zeros((N, 9))
    S = np.zeros((N, 3))

    # Create r+ and r- coordinates
    temp = np.hstack((r2, r1, r3, r1, r3, r2))
    r = np.tile(temp, (N, 1))

    # Create l coordinates
    normabsl1 = np.sqrt(np.sum((r2 - r1) ** 2))
    normabsl2 = np.sqrt(np.sum((r3 - r1) ** 2))
    normabsl3 = np.sqrt(np.sum((r3 - r2) ** 2))
    temp0 = np.hstack(
        ((r2 - r1) / normabsl1, (r3 - r1) / normabsl2, (r3 - r2) / normabsl3)
    )
    l = np.tile(temp0, (N, 1))

    # create unit normal to the edges of the triangle
    u = np.zeros(9)
    u[0:3] = np.cross((r2 - r1) / normabsl1, normal)
    u[3:6] = -np.cross((r3 - r1) / normabsl2, normal)
    u[6:9] = np.cross((r3 - r2) / normabsl3, normal)
    u = np.tile(u, (N, 1))
    # Create projection vector of the observation point
    NORM = np.tile(normal, (N, 1))
    ndot = np.sum(ObsPoint * NORM, axis=1)
    p = ObsPoint - np.tile(ndot.reshape(-1, 1), (1, 3)) * NORM

    # Calculation of the analytic formula
    count = 0
    for c1 in range(3):
        # Distance of observation point perpendicular to the plane with triangle
        temporary = ObsPoint - r[:, 3 * (count + 1) : 3 * (count + 2)]
        distanceobs = np.abs(np.sum(NORM * temporary, axis=1))

        # Calculate p+ and p-
        d1 = np.sum(NORM * r[:, 3 * count : 3 * (count + 1)], axis=1)
        d2 = np.sum(NORM * r[:, 3 * (count + 1) : 3 * (count + 2)], axis=1)

        pplus = r[:, 3 * count : 3 * (count + 1)] - NORM * d1[:, None]
        pminus = r[:, 3 * (count + 1) : 3 * (count + 2)] - NORM * d2[:, None]

        # calculate l+ and l-
        lplus = np.sum(l[:, 3 * c1 : 3 * c1 + 3] * (pplus - p), axis=1)
        lminus = np.sum(l[:, 3 * c1 : 3 * c1 + 3] * (pminus - p), axis=1)

        # Perpendicular distance from projection vector to edge
        P0 = np.abs(np.sum(u[:, 3 * c1 : 3 * c1 + 3] * (pminus - p), axis=1))

        # Distances to l+ and l- from projection vector
        PPLUS = np.sqrt(P0**2 + lplus**2)
        PMINUS = np.sqrt(P0**2 + lminus**2)

        # Vector containing lin P0 measured
        PHAT = (pminus - p - lminus[:, None] * l[:, 3 * c1 : 3 * c1 + 3]) / P0[:, None]

        # Distances to l+ and l- from observation point
        RPLUS = np.sqrt(PPLUS**2 + distanceobs**2)
        RMINUS = np.sqrt(PMINUS**2 + distanceobs**2)
        R0 = np.sqrt(P0**2 + distanceobs**2)

        # A value of one term of the analytic sum 1/R
        d1 = P0 * np.log((RPLUS + lplus) / (RMINUS + lminus))
        d2 = distanceobs * (np.arctan((P0 * lplus) / (R0**2 + distanceobs * RPLUS)))
        d3 = distanceobs * (np.arctan((P0 * lminus) / (R0**2 + distanceobs * RMINUS)))

        dotPHATu = np.sum(PHAT * u[:, 3 * c1 : 3 * c1 + 3], axis=1)
        S[:, c1] = (d1 - d2 + d3) * dotPHATu

        # A value of one term of the analytic sum RHO /R
        d1 = P0 * np.log((RPLUS + lplus) / (RMINUS + lminus))
        d2 = lplus * RPLUS - lminus * RMINUS
        d3 = d1 + d2
        Z[:, 3 * c1 : 3 * c1 + 3] = d3[:, None] * u[:, 3 * c1 : 3 * c1 + 3]

        count += 2

    # Contribution is zero when the projection point on the edge (Wilton et al. 1984, p. 279)
    S[np.isnan(S)] = 0
    S[np.isinf(S)] = 0
    Z[np.isnan(Z)] = 0
    Z[np.isinf(Z)] = 0
    I = np.sum(S, axis=1)
    IRho = 0.5 * (Z[:, 0:3] + Z[:, 3:6] + Z[:, 6:9])

    return I, IRho
